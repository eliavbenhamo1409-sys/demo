    } else if (data.length === 0) {
      body = h('tr', null, h('td', { colSpan: columns.length, className: 'sa-table-empty' }, props.empty || 'אין נתונים להצגה'));
    } else {
      body = data.map(function (row, i) {
        var selected = props.selectedId != null && row.id === props.selectedId;
        return h('tr', { key: row.id != null ? row.id : i, className: cx(props.onRowClick && 'is-clickable', selected && 'is-selected'), 'aria-selected': selected || undefined, tabIndex: props.onRowClick ? 0 : undefined,
          onClick: props.onRowClick ? function () { props.onRowClick(row); } : undefined,
          onKeyDown: props.onRowClick ? function (e) { if (e.key === 'Enter') props.onRowClick(row); } : undefined },
          columns.map(function (c) {
            var v = c.render ? c.render(row, i) : (row[c.key] == null ? '' : String(row[c.key]));
            return h('td', { key: String(c.key), className: cx(c.align === 'num' && 'is-num', c.align === 'ltr' && 'is-ltr', c.mono && 'is-mono', c.muted && 'is-muted', c.className) }, v);
          }));
      });
    }
    return h('div', { className: cx('sa-table-wrap', props.className), style: props.style },
      h('table', { className: 'sa-table' }, props.caption ? h('caption', { className: 'sa-sr' }, props.caption) : null, head, h('tbody', null, body)),
      props.footer ? h('div', { className: 'sa-table-caption' }, props.footer) : null);
  }

  /* ── Stat (KPI) ── */
  function Stat(props) {
    var delta = props.delta;
    var dir = delta == null ? null : delta > 0 ? 'up' : delta < 0 ? 'down' : 'flat';
    var good = props.deltaGood === undefined ? (dir === 'up') : (dir === 'up' ? props.deltaGood : dir === 'down' ? !props.deltaGood : null);
    return h('div', { className: cx('sa-stat', props.tone && 'is-' + props.tone, props.className) },
      h('div', { className: 'sa-stat-label' }, h('span', { style: { display: 'inline-flex', alignItems: 'center', gap: '8px' } }, props.tone ? h(Square, { tone: props.tone }) : null, props.label), props.icon ? h(Icon, { name: props.icon }) : null),
      h('div', { className: 'sa-stat-value' }, h('span', null, props.value), props.unit ? h('span', { className: 'sa-stat-unit' }, props.unit) : null),
      h('div', { className: 'sa-stat-foot' },
        dir ? h('span', { className: cx('sa-delta', dir === 'flat' ? 'is-flat' : good ? 'is-up' : 'is-down') }, h('span', { className: 'sa-delta-num' }, dir !== 'flat' ? h(Icon, { name: dir === 'up' ? 'arrow-up' : 'arrow-down', size: 12, strokeWidth: 2.5 }) : null, (delta > 0 ? '+' : '') + delta + (props.deltaUnit && /^[%\w]/.test(props.deltaUnit) ? props.deltaUnit : '')), props.deltaUnit && !/^[%\w]/.test(props.deltaUnit) ? h('span', null, props.deltaUnit.trim()) : null) : null,
        props.note ? h('span', null, props.note) : null));
  }

  /* ── QuantityStepper ── */
  function QuantityStepper(props) {
    var _v = useState(props.defaultValue != null ? props.defaultValue : 1);
    var value = props.value !== undefined ? props.value : _v[0];
    var min = props.min != null ? props.min : 0, max = props.max != null ? props.max : Infinity, step = props.step || 1;
    function set(n) { n = Math.max(min, Math.min(max, n)); _v[1](n); if (props.onChange) props.onChange(n); }
    return h('div', { style: { display: 'inline-flex', alignItems: 'center' } },
      h('div', { className: cx('sa-qty', props.size === 'sm' && 'is-sm', props.className), role: 'group', 'aria-label': props.label || 'כמות' },
        h('button', { type: 'button', 'aria-label': 'הפחת', disabled: value <= min, onClick: function () { set(value - step); } }, h(Icon, { name: 'minus', size: 20, strokeWidth: 2.5 })),
        h('input', { type: 'number', inputMode: 'numeric', value: value, min: min, max: max === Infinity ? undefined : max, step: step, 'aria-label': props.label || 'כמות', onChange: function (e) { var n = parseInt(e.target.value, 10); if (!isNaN(n)) set(n); } }),
        h('button', { type: 'button', 'aria-label': 'הוסף', disabled: value >= max, onClick: function () { set(value + step); } }, h(Icon, { name: 'plus', size: 20, strokeWidth: 2.5 }))),
      props.unit ? h('span', { className: 'sa-qty-unit' }, props.unit) : null);
  }

  /* ── Stepper ── */
  function Stepper(props) {
    var steps = props.steps || [];
    var c = steps.map(function (s) { return s.key; }).indexOf(props.current);
    return h('ol', { className: cx('sa-steps', props.className), style: { listStyle: 'none', margin: 0, padding: 0 } }, steps.map(function (s, i) {
      var status = i < c ? 'done' : i === c ? 'active' : 'pending';
      return h('li', { key: s.key, className: cx('sa-step', 'is-' + status), 'aria-current': status === 'active' ? 'step' : undefined },
        h('span', { className: 'sa-step-dot' }, status === 'done' ? h(Icon, { name: 'check', size: 14, strokeWidth: 3 }) : (status === 'active' && props.busy) ? h(Spinner, { size: 'sm' }) : String(i + 1)),
        h('div', { className: 'sa-step-body' }, h('p', { className: 'sa-step-label' }, s.label), s.hint ? h('p', { className: 'sa-step-hint' }, s.hint) : null, h('div', { className: 'sa-step-bar' }, status === 'active' && props.busy ? h('span') : null)));
    }));
  }
