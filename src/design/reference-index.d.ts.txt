import type * as React from 'react';

export type IconName =
  | 'search' | 'x' | 'check' | 'check-circle' | 'plus' | 'minus' | 'trash' | 'edit' | 'bell' | 'user' | 'log-out'
  | 'chevron-down' | 'chevron-up' | 'chevron-left' | 'chevron-right' | 'arrow-left' | 'arrow-right' | 'arrow-up' | 'arrow-down'
  | 'arrow-down-left' | 'arrow-up-right' | 'alert-triangle' | 'alert-circle' | 'info' | 'package' | 'grid' | 'book-open'
  | 'more-horizontal' | 'filter' | 'clock' | 'calendar' | 'external-link' | 'download' | 'upload' | 'file-text' | 'inbox'
  | 'truck' | 'tag' | 'link' | 'refresh' | 'camera' | 'menu' | 'layers' | 'activity' | 'hash' | 'eye' | 'copy' | 'printer';

export type Tone = 'neutral' | 'success' | 'warning' | 'danger' | 'accent';
export type MovementType = 'in' | 'out' | 'fault' | 'adjust';

/** Feather-style line icon, 24 viewBox, stroke 2, currentColor. */
export interface IconProps { name: IconName; size?: number; strokeWidth?: number; className?: string; style?: React.CSSProperties }
export declare function Icon(props: IconProps): React.ReactElement;

/** Loading ring that inherits the text colour. */
export interface SpinnerProps { size?: 'sm' | 'md' | 'lg'; track?: boolean; label?: string; className?: string }
export declare function Spinner(props: SpinnerProps): React.ReactElement;

/** A solid colour square with a glyph in the on-colour — the one place an icon is coloured. */
export interface MarkProps { tone?: Tone; type?: MovementType; icon?: IconName; size?: 'sm' | 'md' | 'lg'; strokeWidth?: number; className?: string; children?: React.ReactNode }
export declare function Mark(props: MarkProps): React.ReactElement;
/** A bare 10px colour square, always beside a word. */
export interface SquareProps { tone?: Tone; className?: string }
export declare function Square(props: SquareProps): React.ReactElement;

/** Initials of a worker; `inverse` for the current user. */
export interface AvatarProps { name?: string; size?: 'sm' | 'md' | 'lg'; inverse?: boolean; icon?: IconName; className?: string }
export declare function Avatar(props: AvatarProps): React.ReactElement;

/** One button, six roles. `primary` is the near-black action — once per area. */
export interface ButtonProps extends React.ButtonHTMLAttributes<HTMLButtonElement> {
  variant?: 'primary' | 'secondary' | 'ghost' | 'accent' | 'danger' | 'link';
  size?: 'sm' | 'md' | 'lg' | 'touch';
  icon?: IconName; iconEnd?: IconName; iconOnly?: boolean;
  isLoading?: boolean;
  /** Keyboard hint rendered as a kbd chip. */
  kbd?: string;
}
export declare function Button(props: ButtonProps): React.ReactElement;

/** Status tag — one or two words, always with a word or icon, never colour alone. */
export interface BadgeProps { tone?: Tone | 'inverse'; /** Full-colour fill with the on-colour text; default is an outlined badge with a solid colour square. */ solid?: boolean; dot?: boolean; icon?: IconName; count?: boolean; size?: 'md' | 'lg'; className?: string; children?: React.ReactNode }
export declare function Badge(props: BadgeProps): React.ReactElement;

/** Labelled text field; `mono` for SKUs and quantities, `size="touch"` in the worker app. */
export interface InputProps extends Omit<React.InputHTMLAttributes<HTMLInputElement>, 'size'> {
  label?: string; hint?: string; error?: string; icon?: IconName; end?: React.ReactNode;
  size?: 'md' | 'lg' | 'touch'; mono?: boolean; multiline?: boolean; required?: boolean; optional?: boolean; className?: string;
}
export declare function Input(props: InputProps): React.ReactElement;

/** Searchable select; `allLabel` adds an empty "all" option first. */
export interface SelectOption { value: string; label: React.ReactNode }
export interface SelectProps {
  options: (string | SelectOption)[]; value?: string; defaultValue?: string; onChange?: (value: string) => void;
  label?: string; placeholder?: string; allLabel?: string; searchPlaceholder?: string; emptyLabel?: string; searchable?: boolean;
  disabled?: boolean; open?: boolean; defaultOpen?: boolean; inline?: boolean; activeIndex?: number;
  minWidth?: number; maxWidth?: number; block?: boolean; className?: string;
}
export declare function Select(props: SelectProps): React.ReactElement;

/** The one global search (top bar) — also the local table filter with `kbd={false}`. */
export interface SearchBarProps { value?: string; defaultValue?: string; onChange?: (q: string) => void; onSubmit?: (q: string) => void; isSearching?: boolean; placeholder?: string; kbd?: string | false; className?: string }
export declare function SearchBar(props: SearchBarProps): React.ReactElement;

/** Segmented (default) or `line` tabs; `count` renders a Badge per tab. */
export interface TabItem { value: string; label: React.ReactNode; count?: number; icon?: IconName }
export interface TabsProps { tabs: TabItem[]; value?: string; defaultValue?: string; onChange?: (value: string) => void; variant?: 'segmented' | 'line'; className?: string }
export declare function Tabs(props: TabsProps): React.ReactElement;

/** Surface block with optional header (title, count, actions) and footer. */
export interface CardProps { title?: React.ReactNode; subtitle?: React.ReactNode; icon?: IconName; count?: number; actions?: React.ReactNode; footer?: React.ReactNode; flush?: boolean; className?: string; style?: React.CSSProperties; children?: React.ReactNode }
export declare function Card(props: CardProps): React.ReactElement;

/** Dense data table: 44px rows, sticky micro headers, mono numeric columns, selected row, hover row actions. */
export interface TableColumn<T> { key: keyof T | string; title: React.ReactNode; width?: string; align?: 'start' | 'num' | 'ltr'; mono?: boolean; muted?: boolean; className?: string; render?: (row: T, index: number) => React.ReactNode }
export interface TableProps<T extends { id?: string | number } = Record<string, unknown>> {
  columns: TableColumn<T>[]; data: T[]; isLoading?: boolean; skeletonRows?: number; empty?: React.ReactNode;
  onRowClick?: (row: T) => void; selectedId?: string | number; footer?: React.ReactNode; caption?: string; className?: string; style?: React.CSSProperties;
}
export declare function Table<T extends { id?: string | number }>(props: TableProps<T>): React.ReactElement;

/** KPI tile — three per overview. `deltaGood` says whether a positive delta is good. */
export interface StatProps { label: string; value: string | number; unit?: string; icon?: IconName; delta?: number; deltaUnit?: string; deltaGood?: boolean; note?: React.ReactNode; tone?: 'warning' | 'danger'; className?: string }
export declare function Stat(props: StatProps): React.ReactElement;

/** The "needs attention" list of the overview; renders an EmptyState when empty. */
export interface AttentionItem { id?: string | number; severity?: 'danger' | 'warning' | 'neutral'; /** The word in the solid badge: אזל, תקלה, לא מאושר, מינימום. */ label?: string; title: React.ReactNode; meta?: React.ReactNode[]; action?: string; onAction?: () => void; secondary?: string; onSecondary?: () => void }
export interface AttentionProps { items: AttentionItem[]; emptyTitle?: string; emptyHint?: string; className?: string }
export declare function Attention(props: AttentionProps): React.ReactElement;

/** Details drawer (440px). Rendered in flow here; the app fixes it to the viewport over `bg-scrim`. */
export interface SidePanelProps { title: React.ReactNode; eyebrow?: React.ReactNode; subtitle?: React.ReactNode; tabs?: TabItem[]; tab?: string; defaultTab?: string; onTab?: (value: string) => void; headActions?: React.ReactNode; footer?: React.ReactNode; onClose?: () => void; card?: boolean; className?: string; style?: React.CSSProperties; children?: React.ReactNode }
export declare function SidePanel(props: SidePanelProps): React.ReactElement;

/** Definition list of label/value pairs. */
export interface FieldItem { label: React.ReactNode; value?: React.ReactNode; num?: boolean }
export interface FieldsProps { items: FieldItem[]; columns?: 1 | 2 | 3; className?: string }
export declare function Fields(props: FieldsProps): React.ReactElement;

/** Movement history of one product. */
export interface TimelineItem { id?: string | number; type?: MovementType; title?: React.ReactNode; when?: string; by?: string; qty?: number }
export interface TimelineProps { items: TimelineItem[]; className?: string }
export declare function Timeline(props: TimelineProps): React.ReactElement;

/** Inline movement type: icon tile + word. */
export interface MovementProps { type?: MovementType; label?: React.ReactNode; className?: string; children?: React.ReactNode }
export declare function Movement(props: MovementProps): React.ReactElement;

/** Supplier-dictionary mapping state with in-row approve/fix actions. */
export interface MappingStatusProps { status: 'approved' | 'pending'; target?: string; onApprove?: () => void; onFix?: () => void; className?: string }
export declare function MappingStatus(props: MappingStatusProps): React.ReactElement;

/** One row of the right-hand navigation. */
export interface NavItemProps { href?: string; icon?: IconName; active?: boolean; count?: number; onClick?: (e: React.MouseEvent) => void; className?: string; children?: React.ReactNode }
export declare function NavItem(props: NavItemProps): React.ReactElement;

/** The fixed right sidebar: brand, three destinations, user, sign out. */
export interface SidebarItem { key: string; label: string; icon?: IconName; count?: number; href?: string }
export interface SidebarProps { items?: SidebarItem[]; active?: string; onNavigate?: (key: string) => void; brand?: string; brandSub?: string; mark?: React.ReactNode; user?: { name: string; role?: string }; onSignOut?: (e: React.MouseEvent) => void; className?: string; style?: React.CSSProperties }
export declare function Sidebar(props: SidebarProps): React.ReactElement;

/** Top bar: global search, main actions, notifications, user. */
export interface TopBarProps { actions?: React.ReactNode; user?: { name: string }; onSearch?: (q: string) => void; searchPlaceholder?: string; kbd?: string | false; className?: string }
export declare function TopBar(props: TopBarProps): React.ReactElement;

/** The main actions button + menu (receive / issue / fault by default). */
export interface ActionItem { key: string; label: string; sub?: string; icon: IconName; tone?: Tone }
export interface ActionMenuProps { actions?: ActionItem[]; onSelect?: (key: string) => void; label?: string; variant?: ButtonProps['variant']; open?: boolean; defaultOpen?: boolean; inline?: boolean; activeIndex?: number; className?: string }
export declare function ActionMenu(props: ActionMenuProps): React.ReactElement;
export declare const DEFAULT_ACTIONS: ActionItem[];

/** Worker app home tile: received / issued / fault. */
export interface BigActionProps { type?: 'in' | 'out' | 'fault'; title?: string; subtitle?: string; icon?: IconName; onClick?: () => void; className?: string }
export declare function BigAction(props: BigActionProps): React.ReactElement;

/** −/+ quantity control with a large mono number; `size="sm"` inside panels. */
export interface QuantityStepperProps { value?: number; defaultValue?: number; onChange?: (n: number) => void; min?: number; max?: number; step?: number; unit?: string; label?: string; size?: 'md' | 'sm'; className?: string }
export declare function QuantityStepper(props: QuantityStepperProps): React.ReactElement;

/** 2–4 step process indicator. */
export interface StepItem { key: string; label: string; hint?: string }
export interface StepperProps { steps: StepItem[]; current: string; busy?: boolean; className?: string }
export declare function Stepper(props: StepperProps): React.ReactElement;

/** Quiet empty state with up to two actions as children. */
export interface EmptyStateProps { icon?: IconName; title: React.ReactNode; hint?: React.ReactNode; className?: string; children?: React.ReactNode }
export declare function EmptyState(props: EmptyStateProps): React.ReactElement;

/** In-context message; `danger` is role=alert. */
export interface NoticeProps { tone?: Tone; title?: React.ReactNode; icon?: IconName; action?: string; onAction?: () => void; onDismiss?: () => void; className?: string; children?: React.ReactNode }
export declare function Notice(props: NoticeProps): React.ReactElement;

/** Confirmation dialog for irreversible actions. */
export interface ModalProps { title: React.ReactNode; tone?: 'danger' | 'warning'; confirmLabel?: string; cancelLabel?: string; onConfirm?: () => void; onClose?: () => void; busy?: boolean; footerStart?: React.ReactNode; className?: string; children?: React.ReactNode }
export declare function Modal(props: ModalProps): React.ReactElement;

  interface Window {

      Card: typeof Card; Table: typeof Table; Stat: typeof Stat; Attention: typeof Attention; SidePanel: typeof SidePanel; Fields: typeof Fields;
      Timeline: typeof Timeline; Movement: typeof Movement; MappingStatus: typeof MappingStatus; Sidebar: typeof Sidebar; NavItem: typeof NavItem;
      TopBar: typeof TopBar; ActionMenu: typeof ActionMenu; BigAction: typeof BigAction; QuantityStepper: typeof QuantityStepper; Stepper: typeof Stepper;
      EmptyState: typeof EmptyState; Notice: typeof Notice; Modal: typeof Modal; Avatar: typeof Avatar; Spinner: typeof Spinner; Icon: typeof Icon; Mark: typeof Mark; Square: typeof Square;
      DEFAULT_ACTIONS: ActionItem[];
    };
  }
}
